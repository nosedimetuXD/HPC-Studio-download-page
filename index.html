#!/bin/bash
#SBATCH --job-name=qe_silicon_scf
#SBATCH --partition=standard
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=16
#SBATCH --cpus-per-task=1
#SBATCH --time=01:00:00
#SBATCH --output=outputs/slurm-%j.out

# Environment Setup
module purge
module load openmpi/4.1.4 intel/mkl/2022 quantum-espresso/7.3

export OMP_NUM_THREADS=1
export ESPRESSO_TMPDIR="./tmp"

# Execute Quantum ESPRESSO Self-Consistent Field solver
mpirun -np $SLURM_NTASKS pw.x -nk 2 < pw.scf.silicon.in > outputs/pw.scf.silicon.pwo
echo "Job ${SLURM_JOB_ID} completed successfully on $(hostname)"

ecutwfc (REAL) • Namelist &SYSTEM
Kinetic energy cutoff (in Rydberg) for wavefunctions in plane-wave basis. For ultrasoft pseudopotentials or PAW sets, pair with ecutrho >= 8 * ecutwfc.
Default: 0.0 Ry • Mandatory for SCF calculations • Official QE Documentation ↗
